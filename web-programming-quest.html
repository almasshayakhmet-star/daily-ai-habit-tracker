    const scene = new THREE.Scene();
    scene.background = new THREE.Color(0x87ceeb);
    scene.fog = new THREE.Fog(0x87ceeb, 20, 70);

    const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
    camera.position.set(0, 2, 8);

    const renderer = new THREE.WebGLRenderer({ canvas: document.getElementById('game-canvas'), antialias: true });
    renderer.setSize(window.innerWidth, window.innerHeight);
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    renderer.shadowMap.enabled = true;

    const ambientLight = new THREE.AmbientLight(0xffffff, 0.7);
    scene.add(ambientLight);

    const sun = new THREE.DirectionalLight(0xffffff, 0.9);
    sun.position.set(20, 30, 10);
    sun.castShadow = true;
    scene.add(sun);

    const boxGeo = new THREE.BoxGeometry(1, 1, 1);
    const grassMat = new THREE.MeshLambertMaterial({ color: 0x4ade80 });
    const dirtMat = new THREE.MeshLambertMaterial({ color: 0x854d0e });
    const pathMat = new THREE.MeshLambertMaterial({ color: 0xcbd5e1 });
    const woodMat = new THREE.MeshLambertMaterial({ color: 0x8b5a2b });
    const leafMat = new THREE.MeshLambertMaterial({ color: 0x166534 });

    function addBlock(x, y, z, material) {
      const mesh = new THREE.Mesh(boxGeo, material);
      mesh.position.set(x, y, z);
      mesh.castShadow = true;
      mesh.receiveShadow = true;
      scene.add(mesh);
      return mesh;
    }

    for (let x = -32; x < 32; x++) {
      for (let z = -32; z < 32; z++) {
        addBlock(x, 0, z, grassMat);
        addBlock(x, -1, z, dirtMat);
      }
    }

    for (let z = -24; z <= 24; z++) {
      addBlock(0, 1, z, pathMat);
    }

    function createTree(x, z) {
      for (let y = 1; y <= 4; y++) addBlock(x, y, z, woodMat);
      for (let dx = -2; dx <= 2; dx++) {
        for (let dz = -2; dz <= 2; dz++) {
          for (let dy = 3; dy <= 5; dy++) {
            if (Math.abs(dx) + Math.abs(dz) + Math.abs(dy - 4) < 4) {
              addBlock(x + dx, dy, z + dz, leafMat);
            }
          }
        }
      }
    }

    createTree(-10, -12);
    createTree(10, -15);
    createTree(-14, 14);
    createTree(12, 16);

    const stationMaterial = new THREE.MeshLambertMaterial({ color: 0xf59e0b });
    const stationRoofMaterial = new THREE.MeshLambertMaterial({ color: 0x1d4ed8 });
    const lessonStations = [];

    function createStation(x, z, lesson, color = 0xf59e0b) {
      const group = new THREE.Group();
      const baseMat = new THREE.MeshLambertMaterial({ color });

      for (let dx = -1; dx <= 1; dx++) {
        for (let dz = -1; dz <= 1; dz++) {
          const floor = new THREE.Mesh(boxGeo, baseMat);
          floor.position.set(x + dx, 1, z + dz);
          floor.castShadow = true;
          floor.receiveShadow = true;
          group.add(floor);
        }
      }

      const roof = new THREE.Mesh(new THREE.BoxGeometry(3.4, 0.5, 3.4), stationRoofMaterial);
      roof.position.set(x, 3.5, z);
      roof.castShadow = true;
      group.add(roof);

      const beacon = new THREE.Mesh(new THREE.BoxGeometry(0.8, 2, 0.8), new THREE.MeshLambertMaterial({ color: 0xffffff }));
      beacon.position.set(x, 2.5, z);
      beacon.castShadow = true;
      group.add(beacon);

      group.userData.lesson = lesson;
      group.position.set(0, 0, 0);
      scene.add(group);
      lessonStations.push(group);
    }

    createStation(-8, -18, lessons[0], 0xef4444);
    createStation(8, -10, lessons[1], 0xf97316);
    createStation(-8, -2, lessons[2], 0xeab308);
    createStation(8, 6, lessons[3], 0x22c55e);
    createStation(-8, 14, lessons[4], 0x06b6d4);
    createStation(8, 22, lessons[5], 0x8b5cf6);

    const controls = new PointerLockControls(camera, document.body);
    const velocity = new THREE.Vector3();
    const direction = new THREE.Vector3();
    let moveForward = false;
    let moveBackward = false;
    let moveLeft = false;
    let moveRight = false;
    let canJump = false;
    const keys = {};

    const completedLessons = new Set();
    let activeLesson = null;
    let interactionLocked = false;

    const startScreen = document.getElementById('start-screen');
    const lessonModal = document.getElementById('lesson-modal');
    const successModal = document.getElementById('success-modal');
    const interactionPrompt = document.getElementById('interaction-prompt');
    const progressText = document.getElementById('progress-text');
    const lessonTitle = document.getElementById('lesson-title');
    const lessonDescription = document.getElementById('lesson-description');
    const lessonPoints = document.getElementById('lesson-points');
    const lessonCode = document.getElementById('lesson-code');

    function updateProgress() {
      progressText.textContent = `${completedLessons.size} / ${lessons.length} lessons completed`;
      if (completedLessons.size === lessons.length) {
        successModal.classList.remove('hidden');
      }
    }

    function openLesson(lesson) {
      activeLesson = lesson;
      interactionLocked = true;
      lessonTitle.textContent = lesson.title;
      lessonDescription.textContent = lesson.description;
      lessonPoints.innerHTML = lesson.points.map(point => `
  • ${point}
    • `).join('');
      lessonCode.textContent = lesson.code;
      lessonModal.classList.remove('hidden');
      controls.unlock();
    }

    function closeLesson() {
      lessonModal.classList.add('hidden');
      interactionLocked = false;
    }

    function markLessonComplete() {
      if (!activeLesson) return;
      completedLessons.add(activeLesson.id);
      updateProgress();
      closeLesson();
    }

    document.getElementById('btn-start').addEventListener('click', () => {
      startScreen.style.display = 'none';
      controls.lock();
    });

    document.getElementById('btn-close-lesson').addEventListener('click', closeLesson);
    document.getElementById('btn-close-lesson-secondary').addEventListener('click', closeLesson);
    document.getElementById('btn-complete-lesson').addEventListener('click', markLessonComplete);
    document.getElementById('btn-close-success').addEventListener('click', () => successModal.classList.add('hidden'));

    document.addEventListener('keydown', (e) => {
      keys[e.code] = true;
      switch (e.code) {
        case 'ArrowUp':
        case 'KeyW':
          moveForward = true;
          break;
        case 'ArrowLeft':
        case 'KeyA':
          moveLeft = true;
          break;
        case 'ArrowDown':
        case 'KeyS':
          moveBackward = true;
          break;
        case 'ArrowRight':
        case 'KeyD':
          moveRight = true;
          break;
        case 'Space':
          if (canJump && controls.isLocked) {
            velocity.y += 12;
            canJump = false;
          }
          break;
      }
    });

    document.addEventListener('keyup', (e) => {
      keys[e.code] = false;
      switch (e.code) {
        case 'ArrowUp':
        case 'KeyW':
          moveForward = false;
          break;
        case 'ArrowLeft':
        case 'KeyA':
          moveLeft = false;
          break;
        case 'ArrowDown':
        case 'KeyS':
          moveBackward = false;
          break;
        case 'ArrowRight':
        case 'KeyD':
          moveRight = false;
          break;
      }
    });

    const raycaster = new THREE.Raycaster();
    const center = new THREE.Vector2(0, 0);

    function checkInteraction() {
      if (interactionLocked) {
        interactionPrompt.style.opacity = 0;
        return;
      }

      raycaster.setFromCamera(center, camera);
      const stationMeshes = lessonStations.flatMap((group) => group.children);
      const intersects = raycaster.intersectObjects(stationMeshes, true);

      if (intersects.length > 0 && intersects[0].distance < 6) {
        interactionPrompt.style.opacity = 1;
        if (keys['KeyE']) {
          const parentGroup = lessonStations.find(group => group.children.includes(intersects[0].object));
          if (parentGroup) {
            keys['KeyE'] = false;
            openLesson(parentGroup.userData.lesson);
          }
        }
      } else {
        interactionPrompt.style.opacity = 0;
      }
    }

    const clock = new THREE.Clock();

    function animate() {
      requestAnimationFrame(animate);
      const delta = clock.getDelta();

      if (controls.isLocked) {
        velocity.x -= velocity.x * 10.0 * delta;
        velocity.z -= velocity.z * 10.0 * delta;
        velocity.y -= 9.8 * 3.0 * delta;

        direction.z = Number(moveForward) - Number(moveBackward);
        direction.x = Number(moveRight) - Number(moveLeft);
        direction.normalize();

        if (moveForward || moveBackward) velocity.z -= direction.z * 90.0 * delta;
        if (moveLeft || moveRight) velocity.x -= direction.x * 90.0 * delta;

        controls.moveRight(-velocity.x * delta);
        controls.moveForward(-velocity.z * delta);
        camera.position.y += velocity.y * delta;

        if (camera.position.y < 2) {
          velocity.y = 0;
          camera.position.y = 2;
          canJump = true;
        }

        checkInteraction();
      }

      lessonStations.forEach((group, index) => {
        group.rotation.y = Math.sin(Date.now() * 0.001 + index) * 0.08;
      });

      renderer.render(scene, camera);
    }

    window.addEventListener('resize', () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
    });

    updateProgress();
    animate();
